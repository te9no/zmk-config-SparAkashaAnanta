cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(ananta_scan_startup)
if(NOT DEFINED FIXED OR NOT DEFINED ZMK_APP_SRC)
  message(FATAL_ERROR "Pass FIXED=0/1 and pinned ZMK_APP_SRC")
endif()
execute_process(COMMAND ${PYTHON_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py
  ${ZMK_APP_SRC} ${CMAKE_CURRENT_BINARY_DIR} ${FIXED} RESULT_VARIABLE result)
if(NOT result EQUAL 0)
  message(FATAL_ERROR "Source extraction failed")
endif()
target_include_directories(app PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(app PRIVATE FIXED=${FIXED})
target_sources(app PRIVATE main.c)
