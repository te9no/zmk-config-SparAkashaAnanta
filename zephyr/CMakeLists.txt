# Generate a keyboard-local guarded driver; never edit the shared west module.
if(CONFIG_PMW3610)
  get_target_property(saa_sources app SOURCES)
  set(saa_pmw_found FALSE)
  foreach(source IN LISTS saa_sources)
    if(source MATCHES "zmk-driver-pmw3610-with-custom-studio-rpc/src/pmw3610.c$")
      set(saa_pmw_found TRUE)
      set(patched_pmw "${CMAKE_CURRENT_BINARY_DIR}/saa_pmw3610.c")
      set(guard_script "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/patch_pmw_guard.py")
      execute_process(COMMAND ${PYTHON_EXECUTABLE} ${guard_script} ${source} ${patched_pmw}
                      RESULT_VARIABLE patch_result)
      if(NOT patch_result EQUAL 0)
        message(FATAL_ERROR "Ananta PMW guard: pinned source changed; review required")
      endif()
      get_filename_component(pmw_src_dir ${source} DIRECTORY)
      target_include_directories(app PRIVATE ${pmw_src_dir})
      list(REMOVE_ITEM saa_sources ${source})
      list(APPEND saa_sources ${patched_pmw})
      set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${source} ${guard_script})
    endif()
  endforeach()
  if(NOT saa_pmw_found)
    message(FATAL_ERROR "Ananta PMW source not found; refusing unguarded build")
  endif()
  set_property(TARGET app PROPERTY SOURCES ${saa_sources})
endif()
